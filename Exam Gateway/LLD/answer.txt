from abc import ABC, abstractmethod
from typing import List

class NoNotifierError(Exception): pass

# Strategy Interface
class INotifier(ABC):
    @abstractmethod
    def send(self, message: str) -> str: 
        pass

# Concrete Strategies (Leafs)
class Email(INotifier):
    def send(self, message: str) -> str:
        return f"[EMAIL] {message}"

class SMS(INotifier):
    def send(self, message: str) -> str:
        return f"[SMS] {message}"
    
class SlackPush(INotifier):
    def send(self, message: str) -> str:
        return f"[SLACK PUSH] {message}"

# Composite Pattern Implementation
class NotifierComposition(INotifier):
    def __init__(self):
        self.notifiers: List[INotifier] = []

    def subscribe(self, notifier: INotifier):
        self.notifiers.append(notifier)
    
    def remove(self, notifier: INotifier):
        self.notifiers.remove(notifier)  # Fixed typo

    def send(self, message: str) -> str:
        if not self.notifiers:
            raise NoNotifierError("User has no notifier!")
        
        results = [notifier.send(message) for notifier in self.notifiers]
        return "\n".join(results)

# Decorator Pattern Implementation
class NotifierDecorator(INotifier, ABC):
    def __init__(self, notifier: INotifier):
        self.wrapped = notifier
    
    def send(self, message: str) -> str:
        return self.wrapped.send(message)

class SMSDecorator(NotifierDecorator):
    def send(self, message: str) -> str:
        base_result = super().send(message)
        return f"{base_result}\n[SMS] {message}"  # Fixed \n escape

class SlackPushDecorator(NotifierDecorator):
    def send(self, message: str) -> str:
        base_result = super().send(message)
        return f"{base_result}\n[SLACK PUSH] {message}"  # Fixed \n escape

# Context / Core Logic
class User:
    def __init__(self, notifier: INotifier):
        self.notifier = notifier

    def publish(self, message: str):
        # Now handles return strings consistently
        print(self.notifier.send(message))


# === Execution ===

print("--- Strategy + Composite Execution ---")
composite = NotifierComposition()
composite.subscribe(Email())
composite.subscribe(SlackPush())

user1 = User(composite)
user1.publish("Message Sent!")

print("\n--- Strategy + Decorator Execution ---")
decorator_chain = SlackPushDecorator(Email())
user2 = User(decorator_chain)
user2.publish("Message Sent!")

from abc import ABC, abstractmethod

# adapter
class JsonConf :
    def get_json(self) :
        return '{"id": "1", "name": "ABC"}'

class XmlConf :
    def get_xml(self) :
        return '<data><id>1</id><name>ABC</name></data>'

class Adapter(ABC) :
    @abstractmethod
    def get_dict(self): pass


class JsonAdapter(Adapter) :
    def __init__(self, json_conf: JsonConf):
        self.json_conf = json_conf
    
    def get_dict(self) :
        data = self.json_conf.get_json()

        # json to dict conversion
        data = {"id": 1, "name": "ABC"}
        return data

class XmlAdapter(Adapter) :
    def __init__(self, xml_conf: XmlConf):
        self.xml_conf = xml_conf
    
    def get_dict(self) :
        data = self.xml_conf.get_xml()

        # xml to dict conversion
        data = {"id": 1, "name": "ABC"}
        return data

# factory
class ConfigurationFactory :
    @staticmethod
    def create_conf_adapter(format_type: str) -> Adapter :
        format_type = format_type.lower()

        if format_type == "json" :
            return JsonAdapter(JsonConf())
        elif format_type == "xml" :
            return XmlAdapter(XmlConf())
        else :
            raise ValueError("Undefined format!")

class Application :
    def __init__(self, adapter: Adapter):
        self.adapter = adapter

    def get_conf(self) :
        conf_dict = self.adapter.get_dict()
        print(f"Format type: {type(conf_dict)}\nConfiguration: {conf_dict}")

from abc import ABC, abstractmethod
import threading
import time

class VendingMachine:
    def __init__(self):
        # 1. Fixed unique state names to prevent variable/method collision
        self.idle_state = IdleState(self)
        self.has_money_state = HasMoneyState(self)
        self.dispensing_state = DispensingState(self)
        self.stock_out_state = OutOfStockState(self)

        self._state = self.idle_state
        self._lock = threading.RLock() # Thread-safe Reentrant Lock
        
        self.inventory = {
            1: {"name": "soda", "stock": 1}
        }
        self.selected_product = None

    def set_state(self, state):
        self._state = state

    # --- Securely Guarded Context Actions ---
    def insert_money(self, amount):
        with self._lock:
            self._state.insert_money(amount)
    
    def select_product(self, product_id):
        with self._lock:
            self._state.select_product(product_id)
    
    def dispense(self):
        with self._lock:
            self._state.dispense()
    
    def cancel_transaction(self):
        with self._lock:
            self._state.cancel_transaction()

    # --- Context Helper Methods ---
    def select_product_by_id(self, product_id):
        return self.inventory.get(product_id)

# ==========================================
# STATE IMPLEMENTATIONS
# ==========================================
class State(ABC):
    @abstractmethod
    def insert_money(self, amount): pass
    
    @abstractmethod
    def select_product(self, product_id): pass
    
    @abstractmethod
    def dispense(self): pass
    
    @abstractmethod
    def cancel_transaction(self): pass

class IdleState(State): 
    def __init__(self, machine: VendingMachine):
        self._machine = machine

    def insert_money(self, amount): 
        print(f"[IDLE] Money Inserted: ${amount}")
        self._machine.set_state(self._machine.has_money_state)
    
    def select_product(self, product_id): 
        print("[IDLE] Error: Insert money before making a selection.")
    
    def dispense(self): 
        print("[IDLE] Error: No payment detected.")
    
    def cancel_transaction(self): 
        print("[IDLE] Error: No active transaction to cancel.")

class HasMoneyState(State): 
    def __init__(self, machine: VendingMachine):
        self._machine = machine
        
    def insert_money(self, amount): 
        print("[HAS_MONEY] Additional money accepted.")
    
    def select_product(self, product_id): 
        product = self._machine.select_product_by_id(product_id)
        if not product or product.get("stock") == 0:
            print(f"[HAS_MONEY] Error: Out of stock!")
            self._machine.set_state(self._machine.stock_out_state)
        else:
            print(f"[HAS_MONEY] Selected {product.get('name')}. Initializing dispatch sequence...")
            self._machine.selected_product = product
            self._machine.set_state(self._machine.dispensing_state)
    
    def dispense(self): 
        print("[HAS_MONEY] Select a product first.")
    
    def cancel_transaction(self): 
        print("[HAS_MONEY] Transaction canceled. Refunding money.")
        self._machine.set_state(self._machine.idle_state)

class DispensingState(State): 
    def __init__(self, machine: VendingMachine):
        self._machine = machine
        
    def insert_money(self, amount): 
        print("[DISPENSING] Rejected: Dispense cycle in progress. Please wait.")
    
    def select_product(self, product_id): 
        print("[DISPENSING] Error: Dispense cycle in progress.")
    
    def dispense(self): 
        product = self._machine.selected_product
        
        # Artificial delay to prove our thread-lock blocks simultaneous race conditions
        time.sleep(0.5) 
        
        if product.get("stock") <= 0:
            print(f"[DISPENSING] Race failure: Item emptied mid-air!")
            self._machine.set_state(self._machine.stock_out_state)
        else:
            product["stock"] -= 1
            print(f"[DISPENSING] Released: {product.get('name')} successfully issued. Remaining stock: {product['stock']}")
            self._machine.selected_product = None
            self._machine.set_state(self._machine.idle_state)
    
    def cancel_transaction(self): 
        print("[DISPENSING] Error: Cannot cancel once dispensing begins.")

class OutOfStockState(State): 
    def __init__(self, machine: VendingMachine):
        self._machine = machine
        
    def insert_money(self, amount): 
        print("[OUT_OF_STOCK] Money inserted! Testing system availability...")
        # 1. Take the money and transition to HasMoneyState to give it a chance
        self._machine.set_state(self._machine.has_money_state)
    
    def select_product(self, product_id): 
        print("[OUT_OF_STOCK] Error: Cannot select product. Insert money first.")
    
    def dispense(self): 
        print("[OUT_OF_STOCK] Error: No transaction active.")
    
    def cancel_transaction(self): 
        print("[OUT_OF_STOCK] Refunding core deposit.")
        self._machine.set_state(self._machine.idle_state)

from abc import ABC, abstractmethod
import heapq
from typing import List, Dict

# ==========================================
# 1. THE CORE ENTITIES (Open for Extension)
# ==========================================
class ParkingSpot(ABC): 
    def __init__(self, floor, spot_number: int):
        self.floor = floor  # References Floor object
        self.spot_number = spot_number
        self.is_free = True

    def __lt__(self, other):
        if self.floor.floor_no == other.floor.floor_no:
            return self.spot_number < other.spot_number
        return self.floor.floor_no < other.floor.floor_no

class CompactSpot(ParkingSpot): pass
class MediumSpot(ParkingSpot): pass
class LargeSpot(ParkingSpot): pass
class ElectricSpot(ParkingSpot): pass  # Added seamlessly!


# ==========================================
# 2. THE SPECIFICATION PATTERN (The OCP Shield)
# ==========================================
class ISpsification(ABC):
    @abstractmethod
    def is_satisfied_by(self, spot: ParkingSpot) -> bool:
        pass

# Concrete Rules encapsulating matching logic completely outside the Parking Lot
class MotorcycleSpecification(ISpsification):
    def is_satisfied_by(self, spot: ParkingSpot) -> bool:
        # Motorcycles can fit into any standard space
        return isinstance(spot, (CompactSpot, MediumSpot, LargeSpot))

class CarSpecification(ISpsification):
    def is_satisfied_by(self, spot: ParkingSpot) -> bool:
        return isinstance(spot, (MediumSpot, LargeSpot))

class TruckSpecification(ISpsification):
    def is_satisfied_by(self, spot: ParkingSpot) -> bool:
        return isinstance(spot, LargeSpot)

class ElectricVehicleSpecification(ISpsification):
    def is_satisfied_by(self, spot: ParkingSpot) -> bool:
        # Strictly requires a dedicated charging space
        return isinstance(spot, ElectricSpot)


# ==========================================
# 3. VEHICLE DOMAIN LAYER
# ==========================================
class Vehicle(ABC): 
    def __init__(self, vehicle_id: str, specification: ISpsification):
        self.vehicle_id = vehicle_id
        self.specification = specification

    def can_fit_in(self, spot: ParkingSpot) -> bool:
        return self.specification.is_satisfied_by(spot)

# Derived variants inject their rules dynamically
class Bike(Vehicle):
    def __init__(self, vehicle_id: str):
        super().__init__(vehicle_id, MotorcycleSpecification())

class Car(Vehicle):
    def __init__(self, vehicle_id: str):
        super().__init__(vehicle_id, CarSpecification())

class Truck(Vehicle):
    def __init__(self, vehicle_id: str):
        super().__init__(vehicle_id, TruckSpecification())

class ElectricCar(Vehicle):
    def __init__(self, vehicle_id: str):
        super().__init__(vehicle_id, ElectricVehicleSpecification())


# ==========================================
# 4. STRUCTURE & ALGORITHMIC ALLOCATION ENGINE
# ==========================================
class Floor: 
    def __init__(self, floor_no: int):
        self.floor_no = floor_no
        # Single consolidated min-heap for absolute spatial closeness optimization
        self.free_spots_heap: List[ParkingSpot] = [] 

    def add_parking_spot(self, parking_spot: ParkingSpot):
        heapq.heappush(self.free_spots_heap, parking_spot)


class ParkingLot: 
    def __init__(self, floors: List[Floor]):
        self.floors = floors

    def park_vehicle(self, vehicle: Vehicle):
        # The allocation engine is now 100% generic code!
        for floor in self.floors:
            temporary_retained_spots = []
            matched_spot = None

            # Scan through the floor's priority heap 
            while floor.free_spots_heap:
                spot = heapq.heappop(floor.free_spots_heap)

                if not spot.is_free:
                    continue  # Stale entry cleanup pass

                # Polymorphic execution via Specification evaluate contract
                if vehicle.can_fit_in(spot):
                    matched_spot = spot
                    break
                else:
                    # Valid free spot, but the current vehicle doesn't fit. 
                    # Hold it safely on the side for a moment.
                    temporary_retained_spots.append(spot)

            # Re-insert non-matching spots back onto the priority heap
            for spot in temporary_retained_spots:
                heapq.heappush(floor.free_spots_heap, spot)

            if matched_spot:
                matched_spot.is_free = False
                print(f"[SUCCESS] {vehicle.vehicle_id} assigned to spot {matched_spot.spot_number} on Floor {floor.floor_no}")
                return f"Ticket ID: {vehicle.vehicle_id} @ {matched_spot.spot_number}"

        print(f"[FAILED] No compliant spatial profile available for vehicle {vehicle.vehicle_id}")
        return None

import time
from typing import List, Dict

# ==========================================
# CUSTOM EXCEPTIONS

class SeatUnavailableException(Exception): 
    pass

class SeatNotFoundException(Exception): 
    pass


# ==========================================
# 1. CORE DOMAIN COMPONENT (Fine-Grained Locking)
# ==========================================
class Seat:
    # State Enum Definitions
    STATE_AVAILABLE = "AVAILABLE"
    STATE_LOCKED = "LOCKED"
    STATE_BOOKED = "BOOKED"

    # Event Enum Definitions
    EVENT_SELECT_SEAT = "SELECT_SEAT"
    EVENT_UNSELECT_SEAT = "UNSELECT_SEAT"
    EVENT_PAYMENT_DONE = "PAYMENT_DONE"
    EVENT_PAYMENT_CANCEL = "PAYMENT_CANCEL"

    def __init__(self, seat_row: str, seat_no: int):
        self.seat_row = seat_row
        self.seat_no = no_str = f"{seat_row}{seat_no}"
        self.seat_no_int = seat_no
        self.current_state = self.STATE_AVAILABLE
        
        # Fine-grained reentrant lock: Only guards this individual spatial slot
        self._lock = threading.RLock()
        
        # Cart hold validation details (5-second threshold for easy test simulation)
        self.lock_duration = 5  
        self.lock_timestamp = None

        # Centralized Table-Driven Transition Matrix
        # Current State -> { Event Trigger -> Output Target State }
        self._transition_matrix = {
            self.STATE_AVAILABLE: {
                self.EVENT_SELECT_SEAT: self.STATE_LOCKED
            },
            self.STATE_LOCKED: {
                self.EVENT_UNSELECT_SEAT: self.STATE_AVAILABLE,
                self.EVENT_PAYMENT_DONE: self.STATE_BOOKED
            },
            self.STATE_BOOKED: {
                self.EVENT_PAYMENT_CANCEL: self.STATE_AVAILABLE
            }
        }

    def _check_lock_expiration(self):
        """Internal helper to automatically evict stale checkout holds upon reading."""
        if self.current_state == self.STATE_LOCKED and self.lock_timestamp:
            if time.time() - self.lock_timestamp >= self.lock_duration:
                print(f"[TIMEOUT] Cart allocation expired on seat {self.seat_row}{self.seat_no_int}")
                self.current_state = self.STATE_AVAILABLE
                self.lock_timestamp = None

    def handle_event(self, event: str):
        with self._lock:
            # Step A: Check if a prior user timed out first
            self._check_lock_expiration()

            # Step B: Scan matrix to confirm if this event is supported right now
            allowed_events = self._transition_matrix.get(self.current_state, {})
            next_state = allowed_events.get(event)

            # Step C: If no entry exists in the table, it's an illegal transition
            if not next_state:
                if self.current_state in (self.STATE_LOCKED, self.STATE_BOOKED) and event == self.EVENT_SELECT_SEAT:
                    raise SeatUnavailableException(
                        f"Seat {self.seat_row}{self.seat_no_int} is currently unavailable (Status: {self.current_state})."
                    )
                print(f"[REJECTED] Action '{event}' is forbidden while seat is '{self.current_state}'.")
                return

            # Step D: Apply transactional metadata updates
            if event == self.EVENT_SELECT_SEAT:
                self.lock_timestamp = time.time()
            elif event in (self.EVENT_UNSELECT_SEAT, self.EVENT_PAYMENT_DONE):
                self.lock_timestamp = None

            # Step E: Commit state shift safely within the lock boundary
            print(f"[MATRIX] Success: {self.current_state} --({event})--> {next_state}")
            self.current_state = next_state

import threading
import heapq
import time
from abc import ABC

# ==========================================
# 1. IMMUTABLE DOMAIN ENCAPSULATION
# ==========================================
class Order(ABC):
    def __init__(self, ticker: str, quantity: int, price: float):
        self.ticker = ticker.upper()
        self.quantity = quantity
        self.price = price
        self.timestamp = time.time()  # Tie-breaker for identical prices (FIFO)

class BuyOrder(Order):
    # Python heaps are Min-Heaps by default. 
    # To make a Max-Heap for Buy Orders, we invert the price comparison logic.
    def __lt__(self, other):
        if self.price == other.price:
            return self.timestamp < other.timestamp
        return self.price > other.price  # Higher price has higher priority

class SellOrder(Order):
    def __lt__(self, other):
        if self.price == other.price:
            return self.timestamp < other.timestamp
        return self.price < other.price  # Lower price has higher priority


# ==========================================
# 2. SEGMENTED TICKER BOOK (Granular Locking)
# ==========================================
class SingleTickerOrderBook:
    """Manages the heaps and matching sequence for a SINGLE specific stock ticker."""
    def __init__(self, ticker: str):
        self.ticker = ticker
        self.buy_heap = []   # Max-Heap (Highest bids on top)
        self.sell_heap = []  # Min-Heap (Lowest asks on top)
        self.lock = threading.Lock() # Guarding ONLY this ticker's book

    def add_order(self, order: Order):
        with self.lock:
            if isinstance(order, BuyOrder):
                heapq.heappush(self.buy_heap, order)
            elif isinstance(order, SellOrder):
                heapq.heappush(self.sell_heap, order)
            
            # Immediately attempt to cross orders after insertion
            self._match_orders()

    def _match_orders(self):
        """Internal execution loop matching best Bids against best Asks."""
        # O(1) Check: Run while the best buy price >= best sell price
        while self.buy_heap and self.sell_heap:
            best_buy = self.buy_heap[0]
            best_sell = self.sell_heap[0]

            # Price crossing condition check
            if best_buy.price >= best_sell.price:
                # Determine how much volume can be cleared atomically
                match_quantity = min(best_buy.quantity, best_sell.quantity)
                execution_price = best_sell.price  # Standard market matching rule

                print(f"[TRADE EXECUTION] Ticker: {self.ticker} | Quantity: {match_quantity} @ ${execution_price:.2f}")

                best_buy.quantity -= match_quantity
                best_sell.quantity -= match_quantity

                # Clean out spent elements from the top of the heaps in O(log N) time
                if best_buy.quantity == 0:
                    heapq.heappop(self.buy_heap)
                if best_sell.quantity == 0:
                    heapq.heappop(self.sell_heap)
            else:
                break # Book is spread apart evenly; no matches possible right now


# ==========================================
# 3. GLOBAL ENGINE CROSS-REFERENCE (O(1) Map)
# ==========================================
class MatchingEngine:
    def __init__(self):
        self._books: dict[str, SingleTickerOrderBook] = {}
        self._global_lock = threading.Lock()

    def _get_or_create_book(self, ticker: str) -> SingleTickerOrderBook:
        """Double-checked locking layout to dynamically initialize books safely."""
        ticker = ticker.upper()
        if ticker not in self._books:
            with self._global_lock:
                if ticker not in self._books:
                    self._books[ticker] = SingleTickerOrderBook(ticker)
        return self._books[ticker]

    def submit_order(self, order: Order):
        """Asynchronous entry gateway for concurrent worker threads."""
        book = self._get_or_create_book(order.ticker)
        book.add_order(order)
