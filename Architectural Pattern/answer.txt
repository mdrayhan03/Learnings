        self._mongo_db: MongoDatabase = mongo_db
        
    def send(self, user_id: int, message: str) -> None:

from typing import NamedTuple

# =====================================================================
# 1. IMMUTABLE STATE
# =====================================================================
# Using NamedTuple ensures the state properties cannot be edited in-place.
# To change state, we MUST create a brand-new snapshot.
class ThemeState(NamedTuple):
    mode: str  # "LIGHT", "DARK", or "HIGH_CONTRAST"


# =====================================================================
# 2. INTENTS (The Actions Framework)
# =====================================================================
class ThemeIntent:
    """Base class for all user intentions."""
    pass

class ToggleDarkModeIntent(ThemeIntent):
    """Represents a user wanting to switch between light and dark modes."""
    pass

class SetAccessibilityModeIntent(ThemeIntent):
    """Represents a user wanting to enforce high contrast."""
    pass


# =====================================================================
# 3. MODEL / CENTRALIZED STORE DISPATCHER
# =====================================================================
class ThemeStore:
    def __init__(self, initial_state: ThemeState, view: 'ThemeView'):
        self._state = initial_state
        self._view = view
        
        # Render the initial setup immediately
        self._view.render(self._state)

    def dispatch(self, intent: ThemeIntent) -> None:
        """The single centralized state processing dispatcher."""
        print(f"\n[STORE] Action Intercepted: processing {type(intent).__name__}...")
        
        # Pure reducer-like logic: (Current State + Intent) -> New State
        if isinstance(intent, ToggleDarkModeIntent):
            new_mode = "DARK" if self._state.mode == "LIGHT" else "LIGHT"
            self._state = ThemeState(mode=new_mode)
            
        elif isinstance(intent, SetAccessibilityModeIntent):
            self._state = ThemeState(mode="HIGH_CONTRAST")
            
        # Unidirectional flow: Push the new immutable snapshot to the view layer
        self._view.render(self._state)


# =====================================================================
# 4. VIEW (Renders Pure State Snapshots)
# =====================================================================
class ThemeView:
    def render(self, state: ThemeState) -> None:
        """Takes a pure, read-only snapshot of state and draws it."""
        print(f"--- [SCREEN DISPLAY] Rendering UI with {state.mode} background theme ---")


# =====================================================================
# MVI FLOW EXECUTION SIMULATION
# =====================================================================
if __name__ == "__main__":
    print("--- Initializing MVI Architecture Cycle ---")
    
    # Initialize components
    view = ThemeView()
    starting_state = ThemeState(mode="LIGHT")
    
    # Store coordinates state transitions exclusively
    store = ThemeStore(initial_state=starting_state, view=view)

    print("\n--- Simulating User Inputs (Intents) ---")

    # User action 1: Toggle theme
    store.dispatch(ToggleDarkModeIntent())

    # User action 2: Enforce high contrast
    store.dispatch(SetAccessibilityModeIntent())

    # User action 3: Toggle again (should bounce out of high contrast back into standard logic)
    store.dispatch(ToggleDarkModeIntent())
